# IMPORTANT — two-step install required:

#   refuses to install both at once, so transformers must be force-upgraded
#   AFTER the first install, bypassing the stale metadata check.
#
# Setup:
#   python -m venv .venv-train
#   source .venv-train/bin/activate
#   # Install torch first via the Unsloth platform-specific command:
#   #   https://docs.unsloth.ai  (selects the right CUDA wheel)
#
#   # Step 1 — install everything; unsloth will pull transformers==5.5.0
#   pip install -r requirements-train.txt
#
#   # Step 2 — force-upgrade transformers, skipping the broken metadata check
#   pip install --no-deps transformers==5.5.2
#
# Run:
#   python -m train --config config.json
# ──────────────────────────────────────────────────────────────────────────────
