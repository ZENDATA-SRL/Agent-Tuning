# ── vLLM serving environment ──────────────────────────────────────────────────
#
# Minimal requirements to serve an exported 16-bit merged model with vLLM.
# vLLM applies BNB 4-bit quantisation at load time (--quantization bitsandbytes),
# achieving the same VRAM footprint as training without pre-quantised weights.
#
# vLLM 0.21.0 pins torch==2.11.0 exactly — incompatible with unsloth
# (requires torch<2.11.0).  Export and serving MUST use separate venvs.
#
# Setup:
#   source .serving-venv/bin/activate
#   pip install -r requirements-vllm.txt
#
# Serve:
#   vllm serve outputs/sft-gemma4-e2b-merged \
#       --quantization bitsandbytes \
#       --load-format bitsandbytes \
#       --gpu-memory-utilization 0.90 \
#       --max-model-len 4096 \
#       --enable-auto-tool-choice \
#       --tool-call-parser pythonic \
#       --host 0.0.0.0 --port 8000
# ──────────────────────────────────────────────────────────────────────────────

# torch==2.11.0, torchvision==0.26.0, torchaudio==2.11.0 are pulled
# automatically by vLLM at the exact versions it requires.
vllm==0.21.0
